    /** CLI 起始行（含 project/session 后缀；与 done 行对称） */
    export function logCliStart(argv: string[]): void {
      runtimeLog('info', 'cli start: ' + argv.join(' ') + cliMetaSuffix());
    }
    /** CLI 完成行（一次性旗标防双写；done 行含 duration_ms） */
    export function logCliDone(exitCode: number): void {
      if (cliDoneLogged) { return; }  // 一次性：正常路径已写 → exit 兜底 / 信号路径不重复
      cliDoneLogged = true;
      runtimeLog('info', 'cli done exit=' + exitCode + ' [duration_ms=' + (Date.now() - PROCESS_START_MS) + ']' + cliMetaSuffix());
    }
    /** 测试用：重置 done 旗标（模块态隔离） */
    export function resetDoneFlagForTest(): void { cliDoneLogged = false; }

    - 模块级 `let cliDoneLogged = false;`（**置位先于写入**——即使日志被 `--no-log` / `OPENFEEL_LOG=0` 关闭也置位，保「done 事件已发生」语义）。
  - （c）新增 `installExitHooks`（**安装点 = bin 单一咽喉**；库侧不自动安装；**不设 VITEST/env 守卫**）：
    ```ts
    /** 安装进程退出兜底（仅 bin 调用）：覆盖 process.exit / 正常结束 / 未捕获异常 / SIGINT / SIGTERM */
    export function installExitHooks(proc: Pick<NodeJS.Process, 'on' | 'once'> = process): void {
      // exit 单源兜底：正常路径已写则由旗标跳过；异常 / 中断 / process.exit 路径由此补齐
      proc.on('exit', (code) => { logCliDone(typeof code === 'number' ? code : (process.exitCode ?? 0)); });
      // 信号：保中断语义（130=SIGINT / 143=SIGTERM）；写行由 exit 钩子单源承担
      proc.once('SIGINT', () => { process.exit(130); });
      proc.once('SIGTERM', () => { process.exit(143); });
    }
    ```
    - 钩子体内全 try/catch 吞异常（best-effort 既有语义；R-86-2 缓解）。

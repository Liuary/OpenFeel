### 步骤 2 — 脚本全文（保存为 `$TMP/e2e.sh` 后以 `bash "$TMP/e2e.sh" "$TMP"` 运行；`$TMP` = 步骤 1 的单一临时根）[REV-77-014 修订]

```bash
#!/usr/bin/env bash
# stage-77 op-002：fast 隔离端到端验收（隔离 HOME + 临时项目；全程不触真实 HOME）
set -u
# [REV-77-014 修订] 单一临时根由调用方（步骤 1）创建并经 $1 传入（消除双 mktemp -d）
TMP="${1:?用法: bash e2e.sh <单一临时根（外层 mktemp -d 创建、位于任何 git 仓库之外）>}"
REPO="/c/Users/Liuary/Dev/Mine/AI/OpenFeel"
of() { node "$REPO/bin/openfeel.js" "$@"; }
PASS=0; FAIL=0
ck() { if [ "$2" = "$3" ]; then echo "PASS [$1] rc=$3"; PASS=$((PASS+1)); else echo "FAIL [$1] expected=$2 got=$3"; FAIL=$((FAIL+1)); fi; }
ckj() { if [ "$2" = "true" ]; then echo "PASS [$1]"; PASS=$((PASS+1)); else echo "FAIL [$1] $3"; FAIL=$((FAIL+1)); fi; }

REAL_HOME="$HOME"
# [REV-77-010 修订] 两级面：① 严格面（CLI 实际可写面）全量清单 + sha256
snap_strict() {
  for d in "$REAL_HOME/.openfeel" "$REAL_HOME/.config/openfeel"; do
    if [ -e "$d" ]; then
      echo "== $d =="
      find "$d" -type f -print0 2>/dev/null | sort -z | xargs -0 sha256sum 2>/dev/null | sort
    else
      echo "== $d ABSENT =="
    fi
  done
}
# ② 宽松面：仅框架受管路径子集（~/.claude 会话期易变路径不在比对范围；排除项登记于报告）
snap_framework() {
  for p in "$REAL_HOME/.claude/agents" "$REAL_HOME/.claude/skills" "$REAL_HOME/.claude/CLAUDE.md" "$REAL_HOME/.claude/settings.json" \
           "$REAL_HOME/.config/opencode/agents" "$REAL_HOME/.config/opencode/skills" "$REAL_HOME/.config/opencode/AGENTS.md" "$REAL_HOME/.config/opencode/opencode.jsonc"; do
    if [ -d "$p" ]; then
      find "$p" -type f -print0 2>/dev/null | sort -z | xargs -0 sha256sum 2>/dev/null | sort
    elif [ -f "$p" ]; then
      sha256sum "$p"
    else
      echo "$p ABSENT"
    fi
  done
}
snap_strict > /tmp/s77-home-before-strict.txt
snap_framework > /tmp/s77-home-before-framework.txt

if git -C "$TMP" rev-parse --show-toplevel >/dev/null 2>&1; then
  echo "FATAL: 临时目录位于 git 仓库内 -> 中止（防 autoCommitOnDone 误提交父仓库；请 rm -rf \"$TMP\" 后更换目录）"; exit 1
fi
PROJ="$TMP/proj"; ISO="$TMP/home"; mkdir -p "$PROJ" "$ISO"
export HOME="$ISO" USERPROFILE="$ISO" XDG_CONFIG_HOME="$ISO/.config"
cd "$PROJ"

# ---- A. 初始化与模式配置 ----
of init . --workspace-only --non-interactive >/dev/null 2>&1; ck "A1 init" 0 $?
of config effective workflow_mode | grep -q "fast"; ck "A2 effective=fast(config.yaml)" 0 $?
of config set defaults.workflow_mode fast >/dev/null 2>&1; ck "A3 config set 同值 no-op" 0 $?

# ---- B. 新建阶段（骨架 + 默认值登记） ----
of plan stage add stage-01 >/dev/null 2>&1; ck "B1 plan stage add" 0 $?
grep -q "工作流模式" .openfeel/plan/v1/stage-01/status.md; ck "B2 骨架含工作流模式行" 0 $?
grep -q "enabled" .openfeel/plan/v1/stage-01/status.md; ck "B3 默认 auto_advance enabled" 0 $?

# ---- C. flow mode --json 字段与来源 ----
of flow mode --stage stage-01 --json > "$TMP/mode1.json" 2>/dev/null
R=$(node -e 'const j=require(process.argv[1]);console.log(j.schemaVersion===1 && String(j.stage).endsWith("stage-01") && j.mode==="fast" && j.source==="status.md" && !!j.modes && !!j.modes.full && !!j.modes.fast)' "$TMP/mode1.json")
ckj "C1 flow mode json=fast/status.md" "$R" "$(head -6 "$TMP/mode1.json" | tr -d '\r' | tr '\n' ' ')"

# ---- D. fast 快线链（单步；每步独立命令） ----
of flow advance --stage stage-01 --to plan_review   >/dev/null 2>&1; ck "D1 plan_pending->plan_review" 0 $?
of flow advance --stage stage-01 --to plan_passed   >/dev/null 2>&1; ck "D2 plan_review->plan_passed" 0 $?
of flow advance --stage stage-01 --to scheme_pending >/dev/null 2>&1; ck "D3 拒绝 scheme_pending" 1 $?
of flow advance --stage stage-01 --to exec_running  >/dev/null 2>&1; ck "D4 ->exec_running" 0 $?
of flow advance --stage stage-01 --to review_pending >/dev/null 2>&1; ck "D5 ->review_pending" 0 $?
of flow advance --stage stage-01 --to review_failed  >/dev/null 2>&1; ck "D6 ->review_failed" 0 $?
of flow advance --stage stage-01 --to review_pending >/dev/null 2>&1; ck "D7 review_failed->review_pending(回环)" 0 $?
of flow advance --stage stage-01 --to review_passed  >/dev/null 2>&1; ck "D8 ->review_passed" 0 $?
of flow advance --stage stage-01 --to test_pending   >/dev/null 2>&1; ck "D9 拒绝 test_pending" 1 $?
of flow advance --stage stage-01 --to archiving      >/dev/null 2>&1; ck "D10 ->archiving" 0 $?
of flow advance --stage stage-01 --to done > "$TMP/done.log" 2>&1; ck "D11 ->done" 0 $?
grep -qi "git" "$TMP/done.log" && echo "INFO [D11 done 输出含 git 跳过语（非 git 仓库）]" || echo "INFO [D11 输出未见 git 语—复核输出]"

# ---- E. 多步自动逐步语义（D-1 复核：经停 plan_review，而非拒绝；[REV-77-009 修订]） ----
of plan stage add stage-02 >/dev/null 2>&1; ck "E1 stage-02 add" 0 $?
of flow advance --stage stage-02 --to plan_passed > "$TMP/multistep.log" 2>&1; ck "E2 plan_pending->plan_passed（期望 rc=0，多步自动逐步）" 0 $?
[ "$(grep -c "已推进：" "$TMP/multistep.log")" = "2" ]; ck "E3 两跳输出（全角「已推进：」×2，经停 plan_review）" 0 $?
grep -q "plan_pending → plan_review" "$TMP/multistep.log" && grep -q "plan_review → plan_passed" "$TMP/multistep.log"; ck "E4 两跳具体串各在（plan_pending → plan_review；plan_review → plan_passed）" 0 $?
grep -q "✓ 已推进:" "$TMP/multistep.log" && echo "INFO [E 汇总行「✓ 已推进:」（半角，第 3 行）在位]" || echo "INFO [E 未见汇总行—复核输出]"

# ---- F. 单阶段切回 full ----
of stage set stage-02 --workflow-mode full >/dev/null 2>&1; ck "F1 stage set full" 0 $?
of flow mode --stage stage-02 --json > "$TMP/mode2.json" 2>/dev/null
R=$(node -e 'const j=require(process.argv[1]);console.log(j.mode==="full" && j.source==="status.md")' "$TMP/mode2.json")
ckj "F2 flow mode=full/status.md" "$R" "$(head -6 "$TMP/mode2.json" | tr -d '\r' | tr '\n' ' ')"
of flow advance --stage stage-02 --to scheme_pending >/dev/null 2>&1; ck "F3 full 恢复：->scheme_pending" 0 $?

# ---- G. 非法值拒绝 + 零写盘 ----
SHA1=$(sha256sum .openfeel/plan/v1/stage-02/status.md | awk '{print $1}')
of stage set stage-02 --workflow-mode bogus >/dev/null 2>&1; ck "G1 非法值 exit=1" 1 $?
SHA2=$(sha256sum .openfeel/plan/v1/stage-02/status.md | awk '{print $1}')
[ "$SHA1" = "$SHA2" ]; ck "G2 status.md sha256 不变" 0 $?

# ---- H. 存量回落 builtin（REV-77-007 事实复核） ----
of plan stage add stage-03 >/dev/null 2>&1
sed -i '/^  workflow_mode: fast$/d' .openfeel/config.yaml
sed -i '/工作流模式/d' .openfeel/plan/v1/stage-03/status.md
of flow mode --stage stage-03 --json > "$TMP/mode3.json" 2>/dev/null
R=$(node -e 'const j=require(process.argv[1]);console.log(j.mode==="fast" && j.source==="builtin")' "$TMP/mode3.json")
ckj "H1 存量回落 builtin=fast" "$R" "$(head -6 "$TMP/mode3.json" | tr -d '\r' | tr '\n' ' ')"

# ---- I. 隔离断言（真实 HOME 两级零污染；[REV-77-010 修订]） ----
snap_strict > /tmp/s77-home-after-strict.txt
snap_framework > /tmp/s77-home-after-framework.txt
if diff -q /tmp/s77-home-before-strict.txt /tmp/s77-home-after-strict.txt >/dev/null; then
  echo "PASS [I1a 严格面（.openfeel + .config/openfeel）全量 sha256 前后一致]"; PASS=$((PASS+1))
else
  echo "FAIL [I1a 严格面发生变化]"; diff /tmp/s77-home-before-strict.txt /tmp/s77-home-after-strict.txt | head -20; FAIL=$((FAIL+1))
fi
if diff -q /tmp/s77-home-before-framework.txt /tmp/s77-home-after-framework.txt >/dev/null; then
  echo "PASS [I1b 宽松面（.claude / .config/opencode 框架受管路径子集）无变化]"; PASS=$((PASS+1))
else
  echo "FAIL [I1b 宽松面框架受管路径变化]"; diff /tmp/s77-home-before-framework.txt /tmp/s77-home-after-framework.txt | head -20; FAIL=$((FAIL+1))
fi

echo "SUMMARY: PASS=$PASS FAIL=$FAIL"
echo "TMP=$TMP"
```

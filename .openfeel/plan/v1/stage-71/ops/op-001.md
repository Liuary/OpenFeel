export function getClaudeConfigDir(): string {

### T6 — `backup.ts` `computeBackupRel` global 分支 fallback（D-CFG）

**位置**：`backup.ts:60-65`（`if (projectPath) {...}` 之后的 global 分支）。

**before**：
```ts
  const rel = relative(getHomedir(), absPath);
  // 越界防御（T26）：相对化结果逃出 HOME（含 '..' 或绝对路径）时拒绝
  if (rel.startsWith('..') || isAbsolute(rel)) {
    throw new BackupError(absPath, `路径逃出用户主目录（HOME=${getHomedir()}）`);
  }
  return { zone: 'global', rel: join('global', rel) };
```

**after**：
```ts
  const abs = resolve(absPath);
  const rel = relative(getHomedir(), abs);
  // HOME 内：沿用原 global/ 分区（零回归）
  if (!rel.startsWith('..') && !isAbsolute(rel)) {
    return { zone: 'global', rel: join('global', rel) };
  }
  // HOME 外（D-CFG / R-7）：global-external/<sha8>/<规范化尾段>，不再抛错（避免「备份失败→skipped」）
  // sha8 = sha256(绝对路径) 前 8 位十六进制，仅 [0-9a-f]，天然规避盘符冒号与保留字符（跨平台口径定稿）
  const sha8 = createHash('sha256').update(abs).digest('hex').slice(0, 8);
  // 尾段：逐段剥离盘符（Windows `X:`）与前导分隔符，替换文件系统保留字符 [<>:"|?*]，保留目录层级
  const segs = abs
    .replace(/^[A-Za-z]:/, '')
    .split(/[\\/]+/)
    .filter(Boolean)
    .map((seg) => seg.replace(/[<>:"|?*]/g, '_'));
  if (segs.length === 0) {
    segs.push('root'); // 防御：极端根路径（正常源为文件，不会触发）
  }
  return { zone: 'global', rel: join('global-external', sha8, ...segs) };
```

- **`sha8` 口径（定稿）**：`sha256(resolve(absPath)).slice(0, 8)`，输入 = **规范化绝对路径（含盘符）**；输出仅 `[0-9a-f]` → **无冒号/保留字符**，跨平台可复现、可作单层目录名。
- **越界定义不变**：仍以 `relative(getHomedir(), abs)` 是否 `..`/绝对来判定；仅**处置**由抛错改 fallback。
- **project 分支不动**：`projectPath` 越界**仍抛 `BackupError`**（T26 保留）。
- **改动范围**：**仅 1 函数 + 2 测试**（`computeBackupRel` + `backup.test.ts` 两处）；不新增导出、不新增辅助函数。
- 调用方 `backupFileBeforeWrite` 的 `dest = join(getGlobalBackupRootPath(), ts, rel)` + `mkdirSync(dirname(dest), {recursive:true})` 对多层尾段天然成立；`rel.split('\\').join('/')` 归一为 POSIX（manifest 与 `backupRel`）。

```ts
it('D-CFG/R-7：HOME 外路径 → global-external/<sha8>/… fallback（不再抛错）', () => {
  const outside = join(projectDir, '..', 'outside-dcfg.txt'); // 位于 mock HOME 之外
  writeFileSync(outside, 'external', 'utf-8');
  try {
    const r = backupFileBeforeWrite(outside, { command: 'setup' });
    expect(r).not.toBeNull();
    expect(r!.backupRel.startsWith('global-external/')).toBe(true);
    const abs = resolve(outside);
    const sha8 = createHash('sha256').update(abs).digest('hex').slice(0, 8);
    expect(r!.backupRel).toContain(sha8);
    const dest = join(backupRoot(), r!.ts, r!.backupRel);
    expect(existsSync(dest)).toBe(true);
    expect(readFileSync(dest, 'utf-8')).toBe('external');
    const manifest = JSON.parse(readFileSync(join(backupRoot(), r!.ts, 'manifest.json'), 'utf-8'));
    expect(manifest.entries[abs]).toBeDefined();
  } finally {
    rmSync(outside, { force: true });
  }
});
```

## 困难点与决策

| # | 困难点 | 备选方案 | 采用 | 理由 |
|:-:|--------|----------|------|------|
| 1 | `CLAUDE_CONFIG_DIR` 读取时机 | (a) 模块级常量；(b) 函数体内读 env | **(b)** | (a) 致测试不可注入（模块加载即固化）；(b) 保 `vi.stubEnv`/`process.env` 注入可测（R-新6） |
| 2 | HOME 外备份路径 `sha8` 口径 | (a) 直接拼接绝对路径；(b) 父目录哈希 + basename；(c) **绝对路径 sha256 + 逐段规范化尾段** | **(c)** | (a) 含盘符冒号/保留字符，跨平台非法；(b) 丢层级、同名尾段需额外去重；(c) 仅 `[0-9a-f]` 目录名 + 保层级 + 唯一（sha8 含盘符，`C:\foo` 与 `D:\foo` 不撞） |
| 3 | D-CFG 越界定义 | (a) 放宽「HOME 内」判定；(b) 保留判定，仅改处置 | **(b)** | 判定语义不变，只把「抛错」改「fallback」，project 分支零回归 |
| 4 | 尾段保留层级 vs 文件名长度 | (a) 仅 basename；(b) 全路径逐段 | **(b)** | 计划字面「规范化绝对路径尾段」；`CLAUDE_CONFIG_DIR` 场景深度有限，`mkdirSync(recursive)` 天然支持 |

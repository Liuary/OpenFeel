```typescript
  });

  it('en 应返回英文字符串', () => {
    expect(t('common.error', 'en')).toBe('Error');
  });

  it('默认语言为 zh-CN', () => {
    expect(t('common.error')).toBe('错误');
  });

  it('缺失 key 应回退到 key 本身并 warn', () => {

    spy.mockRestore();
  });
});

// 测试用例组2: t() 变量插值
describe('t() - 变量插值', () => {
  it('应替换模板中的 {var} 占位符', () => {

    expect(result).toContain('stage-01');
    expect(result).toContain('exec_running');
  });

  it('未提供的变量应保留占位符', () => {
    const result = t('flow.advance.okTmpl', 'zh-CN', { stage: 'stage-01' });
    expect(result).toContain('{to}');
  });
});

// 测试用例组3: getCliLang() 语言解析
describe('getCliLang()', () => {
  const testDir = join(homedir(), '.openfeel-test-' + Date.now());

  beforeEach(() => {
    mkdirSync(testDir, { recursive: true });
  });

  afterEach(() => {
    if (existsSync(testDir)) {
      rmSync(testDir, { recursive: true, force: true });
    }
  });

  it('无全局配置且无项目 .info.json 时返回 zh-CN', () => {
    // 创建不含 lang 字段的 .info.json
    const openfeelDir = join(testDir, '.openfeel');
    mkdirSync(openfeelDir, { recursive: true });
    writeFileSync(join(openfeelDir, '.info.json'), JSON.stringify({ user: 'test' }), 'utf-8');
    expect(getCliLang(testDir)).toBe('zh-CN');
  });

  it('项目 .info.json lang=en 时返回 en', () => {
    const openfeelDir = join(testDir, '.openfeel');
    writeFileSync(join(openfeelDir, '.info.json'), JSON.stringify({ user: 'test', lang: 'en' }), 'utf-8');
    expect(getCliLang(testDir)).toBe('en');
  });
});
```

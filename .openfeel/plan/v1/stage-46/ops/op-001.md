## 变更目标

新增备份基础设施：① `global-paths.ts` 增全局备份根路径函数；② **新建** `src/core/backup.ts`：`backupFileBeforeWrite`（判存→算备份相对路径→落盘→更新 manifest，返回 `{backupRel, ts}`）+ `{ts}` 生成/幂等缓存（**内联于 `backupFileBeforeWrite`，不单独导出**；以接口签名清单为准）+ `BackupError`；③ **整次备份会话（ts 探测 + 每文件备份 + manifest 读改写）在单一 `withFileLock(globalLockPath('backup'))` 临界区内**（REV-005 消 TOCTOU）。

**需求映射**：裁定 #2（备份位置 `~/.openfeel/backup/{timestamp}/`）+ 裁定 #3（所有「已存在且将被写」文件）+ B1/B2/B3/B8。

> 参见 kb/architecture.md #跨进程并发保护架构（原子写 + 建议性文件锁）、kb/patterns.md #原子写模式 / #建议性文件锁模式（**锁不嵌套**）/ #全局/项目双 state 路由模式 / #控制区标记模式。

## 关键设计决策

1. **路径分区（B1）**：
   - 全局资产：`global/<相对 HOME>`（如 `global/.config/opencode/AGENTS.md`）；
   - 项目资产：`project/<basename(projectPath)>-<sha256(projectPath).slice(0,8)>/<相对 projectPath>`。
   - `manifest.json`：`{ ts, command, createdAt, entries: { [sourceAbsPath]: { backupRel, hash, command, time } } }`（辅助映射；人工还原不依赖）。
2. **`{ts}` 语义（B2）**：`yyyyMMddTHHmmssSSS`（**本地时区**）；**进程内首次生成后缓存**；同毫秒撞名依次 `-2`、`-3`；**绝不覆盖既有备份**。
3. **锁作用域（REV-005，写死）**：`backupFileBeforeWrite` 全程包在 `withFileLock(globalLockPath('backup'), () => {…})` 内——**ts 探测/预留 + mkdir + 备份副本写入 + manifest 读改写**均在该临界区内；不与其他文件锁**嵌套**（见 op-003 对全局 jsonc 锁的处置）。
4. **备份失败（B3）**：**抛出 `BackupError`**（携带 `filePath`/`cause`）供调用方跳过写入；错误原因暴露于 `BackupError.message`，不静默。
5. **写前备份**：调用方须在**写目标之前**调用（op-003 落地）。返回 `null` 表示「目标不存在 → 无需备份」。
6. **测试可复位**：导出 `resetBackupSetCache()`（进程内 `{ts}` 缓存清零，供测试隔离；生产不使用）。

### 改动 2：**新建** `src/core/backup.ts`

```ts
/**
 * 部署覆盖前备份（stage-46）
 * 在「目标已存在且本次将被写入/覆盖」前，把原件复制到全局备份根 ~/.openfeel/backup/{ts}/（保留层级），
 * 并更新当次 manifest.json。整次会话在单一 backup 文件锁临界区内，消除 ts 探测 / manifest 读改写的 TOCTOU。
 */
import { existsSync, mkdirSync, readFileSync } from 'node:fs';
import { basename, dirname, join, relative, resolve } from 'node:path';
import { createHash } from 'node:crypto';
import { homedir } from 'node:os';
import { atomicWriteFileSync } from './fs/atomic-write.js';
import { withFileLock, globalLockPath } from './fs/file-lock.js';
import { getGlobalBackupRootPath } from './global-paths.js';

/** 会触发部署覆盖写的调用方（B4 枚举） */
export type BackupCommand = 'setup' | 'update' | 'init' | 'migrate';

/** 备份结果（调用方据此写 update_infos「备份」条目） */
export interface BackupResult {
  /** 相对 ~/.openfeel/backup/{ts}/ 的备份路径（POSIX 分隔符，供条目显示） */
  backupRel: string;
  /** 本进程的备份集时间戳（yyyyMMddTHHmmssSSS[-N]） */
  ts: string;
}

/** 备份失败（供调用方跳过该文件写入，B3） */
export class BackupError extends Error {
  readonly filePath: string;
  constructor(filePath: string, cause: unknown) {
    super(`备份失败：${filePath}（${cause instanceof Error ? cause.message : String(cause)}）`);
    this.name = 'BackupError';
    this.filePath = filePath;
  }
}

let cachedTs: string | null = null;
/** 测试隔离：清空进程内 {ts} 缓存（生产不使用） */
export function resetBackupSetCache(): void {
  cachedTs = null;
}

/** 生成本地时间戳 yyyyMMddTHHmmssSSS（本地时区） */
function nowStamp(): string {
  const d = new Date();
  const p = (n: number, w = 2) => String(n).padStart(w, '0');
  return `${d.getFullYear()}${p(d.getMonth() + 1)}${p(d.getDate())}T${p(d.getHours())}${p(d.getMinutes())}${p(d.getSeconds())}${p(d.getMilliseconds(), 3)}`;
}

/** 计算备份相对路径（B1：global/<HOME相对> 或 project/<basename>-<hash8>/<项目相对>） */
function computeBackupRel(absPath: string, projectPath?: string): { zone: 'global' | 'project'; rel: string } {
  if (projectPath) {
    const proj = resolve(projectPath);
    const rel = relative(proj, absPath);
    const hash8 = createHash('sha256').update(proj).digest('hex').slice(0, 8);
    return { zone: 'project', rel: join('project', `${basename(proj)}-${hash8}`, rel) };
  }
  const rel = relative(homedir(), absPath);
  return { zone: 'global', rel: join('global', rel) };
}

/** 读改写当次 manifest（须在 backup 锁内调用） */
function updateManifest(ts: string, command: BackupCommand, source: string, backupRel: string): void {
  const manifestPath = join(getGlobalBackupRootPath(), ts, 'manifest.json');
  let manifest: { ts: string; command: BackupCommand; createdAt: string; entries: Record<string, { backupRel: string; hash: string; command: BackupCommand; time: string }> };
  try {
    manifest = JSON.parse(readFileSync(manifestPath, 'utf-8'));
  } catch {
    manifest = { ts, command, createdAt: new Date().toISOString(), entries: {} };
  }
  manifest.entries[source] = {
    backupRel,
    hash: createHash('sha256').update(readFileSync(source)).digest('hex'),
    command,
    time: new Date().toISOString(),
  };
  atomicWriteFileSync(manifestPath, JSON.stringify(manifest, null, 2) + '\n');
}

/**
 * 写前备份（整次会话在单一 backup 锁临界区内，REV-005）。
 * @returns 目标不存在 → null（无需备份）；否则 { backupRel, ts }
 * @throws BackupError 备份失败（调用方须跳过该文件写入，B3）
 */
export function backupFileBeforeWrite(
  absPath: string,
  opts: { command: BackupCommand; projectPath?: string },
): BackupResult | null {
  const source = resolve(absPath);
  if (!existsSync(source)) {
    return null; // 目标不存在 → 新建，无需备份
  }
  return withFileLock(globalLockPath('backup'), () => {
    try {
      // 1. ts 目录探测/预留（撞名加 -N，绝不覆盖）
      if (!cachedTs) {
        const base = nowStamp();
        let candidate = base;
        let n = 2;
        while (existsSync(join(getGlobalBackupRootPath(), candidate))) {
          candidate = `${base}-${n++}`;
        }
        mkdirSync(join(getGlobalBackupRootPath(), candidate), { recursive: true });
        cachedTs = candidate;
      }
      const ts = cachedTs;

      // 2. 计算备份相对路径 + 落盘（原子写）
      const { rel } = computeBackupRel(source, opts.projectPath);
      const dest = join(getGlobalBackupRootPath(), ts, rel);
      mkdirSync(dirname(dest), { recursive: true });
      atomicWriteFileSync(dest, readFileSync(source));

      // 3. manifest 读改写（同一临界区）
      updateManifest(ts, opts.command, source, rel.split('\\').join('/'));

      return { backupRel: rel.split('\\').join('/'), ts };
    } catch (err) {
      throw new BackupError(source, err);
    }
  });
}
```

- **注意**：`withFileLock` 需支持同步回调（`file-lock.ts:122`）；`atomicWriteFileSync` 签名见 `fs/atomic-write.ts:60`（支持 string｜Buffer）。
- **备份失败阈值**：`readFileSync`/`mkdirSync`/`atomicWriteFileSync` 任一异常 → `BackupError`（不吞）。
- **不要**在本模块内写 `update_infos` 条目（职责分离：条目由 op-003 调用方 `appendUpdateInfo('backed', …)` 写）。

## 测试计划（要点；完整用例见 op-005①）

- 不存在 → 返回 `null`，不建 `backup/` 目录。
- 已存在 → 备份成功，`backupRel` 分区正确（global/project）、文件内容与源一致、`manifest.json` 含映射。
- 备份失败（如权限/占用注入）→ 抛 `BackupError`。
- `{ts}` 复用：同进程两次调用同 `ts`；不同毫秒 → 新目录；同毫秒撞名 → `-2`。
- **并发**：两进程同时备份 → 各自独立 `{ts}` 目录，互不覆盖，两份 manifest 各自完整（REV-005）。
- 测试隔离：`vi.mock('node:os', homedir→tmp)`（N4），不触碰真实 `~/.openfeel/`。

## 风险与回滚

| # | 风险 | 缓解/回滚 |
|---|------|-----------|
| 1 | 非原子/无锁致半写（R8） | `atomicWriteFileSync` + `globalLockPath('backup')`；验收并发测试 |
| 2 | 锁嵌套（与全局 jsonc 锁） | 本模块**不自持其他锁**；调用方须锁外调用（op-003 明确） |
| 3 | `{ts}` 撞名覆盖既有备份 | 探测循环 `-N` + **绝不覆盖**；测试断言 |
| 4 | Windows 路径分隔符进 manifest | `rel.split('\\').join('/')` 归一为 POSIX 显示 |
| 5 | 备份失败被吞（R3） | 统一抛 `BackupError`；调用方跳过写入 |
| 6 | 备份累积膨胀（R1） | 本阶段不做清理（B8/非目标）；manual 说明手工清理 |
